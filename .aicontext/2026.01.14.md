# Kite Jump 项目开发快照 (2026-01-14)

## 1. 项目概述

当前阶段完成了核心玩法的重构，重点在于：

1. **寒冷与冲刺系统**：实现了基于高度的动态寒冷增长，以及收集充能的冲刺机制。
2. **相机运镜**：实现了基于玩家状态（爬升/下落）的动态 Offset 和冲刺时的 FOV 缩放。
3. **配置驱动**：所有游戏数值（物理、关卡、生成权重）均提取到了 Config 文件中。
4. **架构优化**：
* `InteractableEntity` + `Actions`：组合模式处理所有道具/障碍。
* `Player` + `PlayerState` + `PlayerStatusUI`：分离了物理实体、数值状态和视觉表现。

## 5. 核心实体逻辑 (Entities)

### 5.1 通用交互实体

**文件：`src/entities/InteractableEntity.ts`**

> 纯粹的容器，行为由 `config.actions` 决定。

```typescript
import Phaser from 'phaser';
import type { IAction, InteractionContext } from '../actions/ActionInterfaces';
import Player from '../entities/Player'; 

export interface EntityConfig {
  texture: string;    
  color?: number;     
  scale?: number;     
  actions: IAction[]; 
}

export default class InteractableEntity extends Phaser.Physics.Arcade.Sprite {
  private actions: IAction[] = [];
  private isInteracted: boolean = false;

  constructor(scene: Phaser.Scene, x: number, y: number, texture: string) {
    super(scene, x, y, texture);
  }

  public get arcadeBody(): Phaser.Physics.Arcade.Body {
    return this.body as Phaser.Physics.Arcade.Body;
  }

  public configure(config: EntityConfig) {
    this.setTexture(config.texture);
    if (config.color !== undefined) this.setTint(config.color);
    if (config.scale !== undefined) this.setScale(config.scale);
    
    this.setAlpha(1);
    this.setVisible(true);
    this.setActive(true);

    if (this.arcadeBody) {
        this.arcadeBody.enable = true;
        this.arcadeBody.setAllowGravity(false);
        this.arcadeBody.setImmovable(true);
    }

    this.actions = config.actions;
    this.isInteracted = false;
  }

  public onHit(player: Player) {
    if (this.isInteracted) return;
    this.isInteracted = true;

    const context: InteractionContext = {
      target: this,
      player: player,
      scene: this.scene
    };

    this.actions.forEach(action => action.execute(context));
  }

  public disable() {
    this.setVisible(false);
    this.setActive(false);
    if (this.arcadeBody) {
        this.arcadeBody.enable = false;
    }
  }
}

```

### 5.2 玩家逻辑

**文件：`src/entities/Player.ts**`

> 负责物理移动、输入处理，并协调 `PlayerState` 和 `PlayerStatusUI`。

```typescript
import Phaser from 'phaser';
import { GameConfig } from '../config/GameConfig';
import { EVENTS, gameEvents } from '../managers/events';
import PlayerState from './PlayerState';
import PlayerStatusUI from '../ui/PlayerStatusUI';

export default class Player extends Phaser.Physics.Arcade.Sprite {
  private cursors: Phaser.Types.Input.Keyboard.CursorKeys;
  private worldWidth: number;
  public playerState: PlayerState;
  private statusUI: PlayerStatusUI;

  public get arcadeBody(): Phaser.Physics.Arcade.Body {
    return this.body as Phaser.Physics.Arcade.Body;
  }

  constructor(scene: Phaser.Scene, x: number, y: number) {
    super(scene, x, y, 'kite');
    this.worldWidth = scene.scale.width * GameConfig.level.worldWidthRatio;

    scene.add.existing(this);
    scene.physics.add.existing(this);

    this.playerState = new PlayerState(this);
    this.statusUI = new PlayerStatusUI(scene, this);
    
    this.arcadeBody.setDamping(true); 
    this.setDragX(GameConfig.player.dragX);
    this.setMaxVelocity(GameConfig.player.moveSpeed, GameConfig.player.maxFlySpeed);
    this.cursors = scene.input.keyboard!.createCursorKeys();
  }

  update() {
    this.playerState.update(this.scene.game.loop.delta);
    this.statusUI.update();

    // 状态影响物理
    const gravityMult = this.playerState.getGravityMultiplier();
    this.arcadeBody.setGravityY(GameConfig.physics.gravity.y * (gravityMult - 1));

    const controlMult = this.playerState.getControlModifier();
    const currentAccel = GameConfig.player.acceleration * controlMult;

    // 冲刺/下落限速逻辑
    if (this.playerState.isDashing) {
        this.setVelocityY(GameConfig.playerState.dashSpeed);
    } else {
        if (this.arcadeBody.velocity.y > GameConfig.player.maxFallSpeed) {
            this.setVelocityY(GameConfig.player.maxFallSpeed);
        }
    }

    // 动态加速度 (ScreenWrap 分支特色)
    const absVerticalSpeed = Math.abs(this.arcadeBody.velocity.y);
    const dynamicAccel = currentAccel + (absVerticalSpeed * GameConfig.player.verticalToHorizontalRatio);

    // 输入与移动
    if (this.cursors.left.isDown) {
      this.setAccelerationX(-dynamicAccel);
      this.setFlipX(true);
    } else if (this.cursors.right.isDown) {
      this.setAccelerationX(dynamicAccel);
      this.setFlipX(false);
    } else {
      this.setAccelerationX(0);
    }

    this.checkScreenWrap();
  }

  private checkScreenWrap() {
    const halfWidth = this.width / 2;
    if (this.x < halfWidth) {
      this.x = halfWidth;
      this.setVelocityX(0);
    } else if (this.x > this.worldWidth - halfWidth) {
      this.x = this.worldWidth - halfWidth;
      this.setVelocityX(0);
    }
  }

  public boost(force: number) {
    const currentVel = this.arcadeBody.velocity.y;
    this.setVelocityY(Math.min(currentVel, force)); // 防止减速
  }

  public die(cause: string) {
    gameEvents.emit(EVENTS.GAME_OVER, cause);
  }

  destroy(fromScene?: boolean) {
      this.statusUI.destroy();
      super.destroy(fromScene);
  }
}

```

### 5.3 玩家状态逻辑

**文件：`src/entities/PlayerState.ts**`

> 纯数值逻辑类，处理寒冷公式和冲刺判定。

```typescript
import Phaser from 'phaser';
import { GameConfig } from '../config/GameConfig';
import Player from './Player';

export default class PlayerState {
  private player: Player;
  private initialY: number;
  
  public coldness: number = 0;      
  public dashEnergy: number = 0;    
  
  public isDashing: boolean = false;
  private dashTimer: number = 0;

  constructor(player: Player) {
    this.player = player;
    this.initialY = player.y;
  }

  update(dt: number) {
    if (this.isDashing) {
      this.dashTimer -= dt;
      if (this.dashTimer <= 0) this.endDash();
      return; 
    }

    // 寒冷计算公式：Base + Factor * (Height / 1000)
    const currentHeightPixels = Math.max(0, this.initialY - this.player.y);
    const heightMeters = currentHeightPixels / GameConfig.level.pixelsPerMeter;

    const growthRate = GameConfig.playerState.coldBaseRate + 
                       (GameConfig.playerState.coldHeightFactor * (heightMeters / 1000));

    const deltaCold = growthRate * (dt / 1000);
    this.coldness += deltaCold;
    this.coldness = Phaser.Math.Clamp(this.coldness, 0, 100);

    if (this.coldness >= GameConfig.playerState.thresholds.icebound) {
        this.player.die("frozen");
    }
  }

  public addDashEnergy(amount: number) {
    this.dashEnergy += amount;
    this.dashEnergy = Phaser.Math.Clamp(this.dashEnergy, 0, 100);
    if (this.dashEnergy >= 100) this.activateDash();
  }

  public activateDash() {
    if (this.dashEnergy < 100 || this.isDashing) return;

    this.isDashing = true;
    this.dashEnergy = 0;
    this.dashTimer = GameConfig.playerState.dashDuration;
    this.coldness = 0; // 清除寒冷
    
    this.player.setVelocityY(GameConfig.playerState.dashSpeed);
  }

  private endDash() {
    this.isDashing = false;
  }

  public getGravityMultiplier(): number {
    if (this.isDashing) return 0;
    let multiplier = 1.0;
    const th = GameConfig.playerState.thresholds;
    const pen = GameConfig.playerState.penalties;

    if (this.coldness >= th.chilly) multiplier += pen.chillyGravity;
    if (this.coldness >= th.frozen) multiplier += pen.frozenGravity;
    if (this.coldness >= th.extreme) multiplier += pen.extremeGravity;
    return multiplier;
  }

  public getControlModifier(): number {
    if (this.isDashing) return 2.0;
    if (this.coldness >= GameConfig.playerState.thresholds.frozen) {
        return 1.0 - GameConfig.playerState.penalties.frozenDrag;
    }
    return 1.0;
  }
}
```
